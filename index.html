Google Sheets Webhook URL
Pegá la URL de tu Apps Script web app (doGet).
Apps Script — pegá esto:
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sh = ss.getSheetByName("Entrenamientos");
    if (!sh) {
      sh = ss.insertSheet("Entrenamientos");
      sh.appendRow(["Fecha","Dia","Ejercicio","Series","Reps","Pesos","Musculos"]);
    }
    sh.appendRow([e.parameter.fecha,e.parameter.dia,
      e.parameter.ejercicio,e.parameter.series,
      e.parameter.reps,e.parameter.pesos,e.parameter.muscs]);
    return ContentService.createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ok:false,error:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
