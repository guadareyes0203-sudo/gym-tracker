function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var tipo = e.parameter.tipo || "ejercicio";
  
  if(tipo === "medidas") {
    var sh = ss.getSheetByName("Medidas") || ss.insertSheet("Medidas");
    if(sh.getLastRow()===0)
      sh.appendRow(["Fecha","Peso","Grasa%","Musculo%","Proteina%",
        "Cintura","Cadera","Pecho","Brazo D","Brazo I",
        "Muslo D","Muslo I","Pantorrilla D","Pantorrilla I"]);
    sh.appendRow([e.parameter.fecha,
      e.parameter.peso,e.parameter.grasa,
      e.parameter.musculo,e.parameter.proteina,
      e.parameter.cintura,e.parameter.cadera,e.parameter.pecho,
      e.parameter.brazo_d,e.parameter.brazo_i,
      e.parameter.muslo_d,e.parameter.muslo_i,
      e.parameter.pantorr_d,e.parameter.pantorr_i]);
  } else {
    var sh = ss.getSheetByName("Entrenamientos") || ss.insertSheet("Entrenamientos");
    if(sh.getLastRow()===0)
      sh.appendRow(["Fecha","Dia","Ejercicio","Series","Reps","Pesos","Musculos"]);
    sh.appendRow([e.parameter.fecha,e.parameter.dia,
      e.parameter.ejercicio,e.parameter.series,
      e.parameter.reps,e.parameter.pesos,e.parameter.muscs]);
  }
  return ContentService.createTextOutput("ok");
}
